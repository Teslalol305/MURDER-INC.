# MURDER INC. demo: one take, played by the MIDemo director on a fixed 1/60 s frame (tools/box/demo/capture.sh).
# One action per line; times in game seconds. Verbs (Source/MurderInc/Flow/MIDemoDirector.cpp):
#   wait <s> | con <console command> | capture <fifo>|off | god 0|1 | secure <objective>
#   face path [pitch] | face angles <yaw> <pitch> | face point <x> <y> <z> | look <yaw> <pitch> [s] | lookat <x> <y> <z> [s]
#   spring <hz> <max deg/s> | walk <x> <y> [sprint] [arrive cm] | route <sprint> <x y>... (plan_path.py)
#   auto 0|1 [head] (fight whatever comes into sight while walking) | fight <limit s> [kills] [head] [calm s]
#   hunt <limit s> [kills] [head] (fight; walk the nav path to the nearest Goon when none is in sight) | hide <label part>...
#   goto <x> <y> [sprint] (nav path from where the player is) | alert <radius m>
#   blow [limit s] (the nearest explosive in front) | grab [limit s] (hold E until a reel is taken)
#   guide [limit s] [sprint 0|1] (walk the route the story guidance draws, to the reel's case or until the step changes)
#   waitstate <title|intro|playing|complete> [limit s] | quit

# --- not captured: the map, the title and the stage settle --------------------------------------------------------
wait 4
con MIVoice clean 1
con MIVoice frequency often
con r.MotionBlur.Amount 0.15

capture @FIFO@

# --- the set-up flow ------------------------------------------------------------------------------------------------

wait 1.2

# --- the run --------------------------------------------------------------------------------------------------------
waitstate playing 20
god 1
secure 1                                  # the Chop Shop and the Cage reels were taken earlier in the run
secure 2
spring 1.3 110 1.7                        # walk view, max turn rate, fight view (slower turns keep the arms steady)
face path -2
auto 1 1
route 0 -3725 875 -3475 875 -3175 675 -2775 525 -2775 425 -1500 150   # down the avenue under the El
alert 60                                  # the Goons on the block come to look
fight 22 3 1 5                            # headshots first
con MIInput reload 1                      # a reload
wait 2.8
con MIInput weapon2 1                     # the NINE
wait 1.4
fight 8 1 1 3
auto 0
con MIInput weapon1 1                     # back to the CHOPPER (two takes crashed blowing a tank with the NINE up)
wait 1.0
lookat -420 -800 60 2.5
blow 5                                    # the propane tanks in the garage
wait 2.5
auto 1 1
goto -490 130
auto 0
lookat -400 100 60 1.5
grab 6                                    # the last reel
wait 1.0
auto 0                                    # no stopping now: the pursuit would hold the run off the truck
goto 3475 2525 1                          # to the lot beside the IPX truck
face point 4130 2970 170                  # then onto its rear doors (the extraction zone is 2.5 m round them)
walk 3480 2930 0 60 8
waitstate complete 20
wait 6
capture off
con MIAudio.Record stop @NAME@
wait 0.5
quit
